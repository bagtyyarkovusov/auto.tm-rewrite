// Throwaway PR-only evidence route. Never shipped or committed under app/.
import { useEffect } from "react";
import { Image } from "react-native";
import { useLocalSearchParams } from "expo-router";
import { StatusBar } from "expo-status-bar";
import { useTranslation } from "react-i18next";
import { WizardSchemas } from "@auto-tm/contracts";

import { WizardLayout } from "@/src/listings/wizard/WizardLayout";
import Step2Photos from "@/src/listings/wizard/Step2Photos";
import type { StagedPhoto } from "@/src/listings/uploadStaging/types";
import { themeStore } from "@/src/theme/themeStore";

const images = [
  require("../../../packages/db/scripts/reviewer-fixtures/camry-1.jpg"),
  require("../../../packages/db/scripts/reviewer-fixtures/camry-2.jpg"),
  require("../../../packages/db/scripts/reviewer-fixtures/camry-3.jpg"),
];
const noop = () => {};

export default function PhotoProof() {
  const params = useLocalSearchParams<{ count?: string; theme?: string }>();
  const count = params.count === "3" ? 3 : 2;
  const theme = params.theme === "dark" ? "dark" : "light";
  const { t, i18n } = useTranslation();
  useEffect(() => {
    const previousLanguage = i18n.language;
    let previousTheme = themeStore.getState().theme;
    let live = true;
    void themeStore.getState().hydrate().then(() => {
      if (!live) return;
      previousTheme = themeStore.getState().theme;
      themeStore.setState({ theme });
      void i18n.changeLanguage("en");
    });
    return () => {
      live = false;
      themeStore.setState({ theme: previousTheme });
      void i18n.changeLanguage(previousLanguage);
    };
  }, [theme]);
  const photos: StagedPhoto[] = images.slice(0, count).map((asset, index) => ({
    photoId: `550e8400-e29b-41d4-a716-${String(100 + index).padStart(12, "0")}`,
    key: `proof/${index}/original.jpg`, localUri: Image.resolveAssetSource(asset).uri,
    state: "uploaded", sortOrder: index, retryCount: 0,
  }));
  const canContinue = WizardSchemas.StepPhotosSchema.safeParse({ photos }).success;
  return <>
    <StatusBar style={theme === "dark" ? "light" : "dark"} />
    <WizardLayout routeTitle={t("sellCar")} stepTitle={t("wizardSteps.photos")} stepNumber={3} stepCount={7}
      onBack={noop} onContinue={noop} onPublish={noop} onClose={noop}
      mode="create" editDetourActive={false} canContinue={canContinue} canPublish={false}
      canGoBack isLastStep={false} saveStatus="saved" saveError={null} onRetrySave={noop} progressPercent={3 / 7 * 100}>
      <Step2Photos photos={photos} onAddPhoto={async () => {}} onRemovePhoto={noop} onReorderPhotos={noop}
        onRetryPhoto={noop} isCompressing={false} isUploading={false} />
    </WizardLayout>
  </>;
}
