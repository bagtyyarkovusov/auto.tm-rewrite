// Temporary native UI fixture. Never committed or shipped.
import { useEffect } from "react";
import { Image, ScrollView, View } from "react-native";
import { useLocalSearchParams } from "expo-router";
import { useTranslation } from "react-i18next";
import { PhotoGallery } from "../../../src/listings/components/PhotoGallery";
import { TabScreen } from "../../../components/navigation/TabScreen";
import { useTabBarSpace } from "../../../components/navigation/tabBarHeight";
import { Text } from "../../../components/ui/text";
import { themeStore } from "../../../src/theme/themeStore";

const photos = [
  require("../../../assets/ui718-proof/camry-1.jpg"),
  require("../../../assets/ui718-proof/camry-2.jpg"),
  require("../../../assets/ui718-proof/camry-3.jpg"),
];
const buildMedia = () => photos.map((asset, index) => {
  const uri = Image.resolveAssetSource(asset).uri.replace(/^http:\/\/[^/]+/, "http://127.0.0.1:8101");
  console.info("UI718_ASSET", uri);
  return { id: `ui718-photo-${index}`, kind: "image" as const, key: `ui718/${index}/original.jpg`, sortOrder: index,
    variants: { thumbnail: uri, list: uri, detail: uri, fullscreen: uri } };
});
export default function NativeProof() {
  const { theme = "light", locale = "ru" } = useLocalSearchParams<{ theme?: string; locale?: string }>();
  const { i18n } = useTranslation();
  const paddingBottom = useTabBarSpace();
  const media = buildMedia();
  useEffect(() => {
    const previousTheme = themeStore.getState().theme;
    const previousLocale = i18n.language;
    let live = true;
    void themeStore.getState().hydrate().then(() => { if (live) { themeStore.setState({ theme: theme === "dark" ? "dark" : "light" }); void i18n.changeLanguage(locale); console.info("UI718_FIXTURE_THEME", theme); } });
    return () => { live = false; themeStore.setState({ theme: previousTheme }); void i18n.changeLanguage(previousLocale); };
  }, [theme, locale]);
  return <TabScreen underTabBar><ScrollView contentContainerStyle={{ paddingBottom }}>
    <Text className="p-4 text-title font-bold text-foreground">UI718 native photo fixture</Text>
    {Array.from({ length: 8 }, (_, index) => <View key={index}>
      <Text className="px-4 py-3 text-body text-foreground">Toyota Camry · local fixture {index + 1}</Text>
      <PhotoGallery media={media} />
    </View>)}
  </ScrollView></TabScreen>;
}
