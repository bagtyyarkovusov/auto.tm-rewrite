import subprocess,time,sys
adb='adb'  # Portable tool path; recorded command/timing body unchanged.
def shell(command):
 return subprocess.run([adb,'-s','emulator-5554','shell',command],check=True,capture_output=True,text=True).stdout
mode=sys.argv[1]
assert mode in ('baseline','feature')
workload=[]
for _ in range(4):
 workload.extend(['input swipe 520 1700 520 1000 700','input swipe 520 1000 520 1700 700'])
workload.extend(['input swipe 120 2260 925 2260 1400','input swipe 925 2260 120 2260 1400'])
# Warm the exact workload once. These frames are excluded by the later reset.
for command in workload:
 shell(command);time.sleep(.25)
shell("am start -a android.intent.action.VIEW -d 'autotm:///ui718-proof?theme=dark&run="+mode+"' tm.auto.app")
time.sleep(5)
subprocess.run([adb,'-s','emulator-5554','exec-out','screencap','-p'],check=True,stdout=open('/tmp/ui718-pair-'+mode+'-start.png','wb'))
open('/tmp/ui718-pair-'+mode+'-reset.txt','w').write(shell('dumpsys gfxinfo tm.auto.app reset'))
for command in workload:
 shell(command);time.sleep(.25)
time.sleep(1)
open('/tmp/ui718-pair-'+mode+'-gfx.txt','w').write(shell('dumpsys gfxinfo tm.auto.app framestats'))
subprocess.run([adb,'-s','emulator-5554','exec-out','screencap','-p'],check=True,stdout=open('/tmp/ui718-pair-'+mode+'-end.png','wb'))
print(mode+' warm pair leg collected')
